#+TITLE: 11 - git+https Cawfile Pull

This example pulls a remote Cawfile bundle into the local Ocawe cache, compiles
it into a cached single =ocawecore= binary, and runs the selected workflow
through that binary.

#+begin_src rcl
struct RemotePullInput
  include JSON::Serializable
  getter prompt : String?
end

struct RemotePullOutput
  include JSON::Serializable
  getter status : String?
end

@[Validate(RemotePullInput, RemotePullOutput)]
workflow "11-git-https-pull" do
  input_type RemotePullInput
  output_type RemotePullOutput

  input_validate Schema::Types.object({
    "prompt" => Schema::Types.optional(Schema::Types.of(String)),
  }, strict: false)
  output_validate Schema::Types.object({
    "status" => Schema::Types.optional(Schema::Types.of(String)),
  }, strict: false)

  exec "git+https://github.com/lefinepro/ocawe/caws/10-acp-agent", runtime: {"git+https"}
end
#+end_src

Manual cache refresh without running the workflow:

#+begin_src bash
ocawe pull git+https://github.com/lefinepro/ocawe/caws/10-acp-agent
ocawe pull git+ssh://github.com/lefinepro/ocawe/caws/10-acp-agent
#+end_src
